🚀 Start the App
Connect First, Then Listen
The app connects to MongoDB before listening. In PowerShell, configure the connection and port, then start the server.
$env:MONGODB_URI = "mongodb://127.0.0.1:27017/test"
$env:PORT = "4000"
npm start
📝 Defaults: MONGODB_URI defaults to mongodb://127.0.0.1:27017/test; PORT defaults to 4000.

src/server.js
Wires up Express, mounts the router, and awaits the database connection before calling listen().
import express from "express";
import connectDatabase from "./config/database.js";
import userRouter from "./routes/users.routes.js";

const app = express();
const port = Number(process.env.PORT) || 4000;

app.use(express.json());
app.use("/", userRouter);

await connectDatabase();
app.listen(port, () => {
  console.log(`Server listening on port ${port}`);
});
src/config/database.js
import mongoose from "mongoose";

const databaseUri = process.env.MONGODB_URI || "mongodb://127.0.0.1:27017/test";

export default async function connectDatabase() {
  await mongoose.connect(databaseUri);
  console.log("Connected to MongoDB");
}
src/models/user.model.js
import mongoose from "mongoose";

const userSchema = new mongoose.Schema({
  name: {
    type: String,
    required: true,
  },
  email: {
    type: String,
    required: true,
  },
  age: {
    type: Number,
    min: 18,
    max: 59,
  },
  isDeleted: {
    type: Boolean,
    default: false,
  },
}, { timestamps: true });

const User = mongoose.model("User", userSchema);

export default User;
📝 { timestamps: true } adds createdAt/updatedAt automatically — no manual timestamp fields needed.
src/services/users.service.js
Pure data-access functions — no req/res here. findUsers excludes soft-deleted rows; update and delete both use $set.
import User from "../models/user.model.js";

export async function findUsers() {
  return User.find({ isDeleted: { $ne: true } }).lean();
}

export async function createUser(userData) {
  return User.create(userData);
}

export async function updateUserById(userId, userData) {
  return User.findByIdAndUpdate(userId, { $set: userData }, {
    new: true,
  });
}

export async function deleteUserById(userId) {
  return User.findByIdAndUpdate(userId, { $set: { isDeleted: true } }, {
    new: true,
  });
}
💡 { new: true } makes findByIdAndUpdate return the updated document instead of the one before the change.
src/controllers/users.controller.js
Translates HTTP requests into service calls and shapes the JSON response. Each handler is wrapped in try/catch.
import {
  createUser,
  deleteUserById,
  findUsers,
  updateUserById,
} from "../services/users.service.js";

export async function listUsers(req, res) {
  try {
    const users = await findUsers();
    res.json({ message: "successfully find all users", data: users });
  } catch (error) {
    res.json({ message: error.message || "Unable to fetch user" });
  }
}

export async function createUserHandler(req, res) {
  try {
    await createUser(req.body.user);
    res.json({ message: "user added successfully" });
  } catch (error) {
    res.json({ message: error.message || "Unable to insert user" });
  }
}

export async function updateUserHandler(req, res) {
  try {
    await updateUserById(req.params.userId, req.body.user);
    res.json({ message: "user updated successfully" });
  } catch (error) {
    res.status(500).json({ message: error.message || "Unable to update user" });
  }
}

export async function deleteUserHandler(req, res) {
  try {
    await deleteUserById(req.params.userId);
    res.json({ message: "user deleted successfully" });
  } catch (error) {
    res.status(500).json({ message: error.message || "Unable to delete user" });
  }
}
src/routes/users.routes.js
import express from "express";
import {
  createUserHandler,
  deleteUserHandler,
  listUsers,
  updateUserHandler,
} from "../controllers/users.controller.js";

const router = express.Router();

router.get("/all-user", listUsers);
router.post("/create-user", createUserHandler);
router.patch("/update-user/:userId", updateUserHandler);
router.delete("/delete-user/:userId", deleteUserHandler);

export default router;
📝 Notes & Gotchas
Don't vs Do
✕ Don't
🚫 app.use(connection())

Treating the DB connection as middleware means the server can start accepting requests before Mongo is even connected.

✓ Do
🔌 await connectDatabase()

Await the connection before calling app.listen(), so the database is guaranteed ready before traffic arrives.

✕ Don't
🚫 Manual timestamp fields

Hand-rolling createdAt/updatedAt on the schema is extra code that Mongoose already gives you for free.

✓ Do
⏱️ { timestamps: true }

One schema option automatically maintains both fields on every save and update.

✕ Don't
🚫 User.findByIdAndDelete()

A hard delete destroys the document permanently — no audit trail, no way to restore a user.

✓ Do
🗑️ Soft delete via isDeleted

DELETE /delete-user/:userId only flips isDeleted to true; findUsers filters those out with { isDeleted: { $ne: true } }.

✕ Don't
🚫 Mixed-up folder naming

Inconsistent names across config/controllers/models make the codebase harder to navigate as it grows.

✓ Do
🗂️ Standardized structure

Keep consistent folder and file names: config, controllers, models, routes, and services.

📋 Existing Routes, Restated

  • GET/all-user — list active users.
  • POST/create-user — create a user. The body must use the existing user wrapper, e.g. {"user":{"name":"Ada Lovelace","email":"ada@example.com","age":30}}.
  • PATCH/update-user/:userId — update a user with the same body shape, e.g. {"user":{"age":31}}. Example URL: /update-user/64a1b2c3d4e5f67890123456.
