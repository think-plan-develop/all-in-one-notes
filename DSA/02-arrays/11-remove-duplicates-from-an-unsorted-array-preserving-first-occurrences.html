remove-duplicates-from-an-unsorted-array.js
const arr = [4, 5, 4, 3, 5, 6];

const seen = new Set();
const result = [];

for (const x of arr) {
  if (!seen.has(x)) {
    seen.add(x);
    result.push(x);
  }
}

console.log(result);

Remember these

  • ⚡ Keep order: Building a new array while checking a Set keeps the original order.
