const arr = [1, 2, 3, 4, 5];

function rotateLeft(a) {
  if (a.length < 2) return;
  const first = a[0];
  for (let i = 0; i < a.length - 1; i++) a[i] = a[i + 1];
  a[a.length - 1] = first;
}

function rotateRight(a) {
  if (a.length < 2) return;
  const last = a[a.length - 1];
  for (let i = a.length - 1; i > 0; i--) a[i] = a[i - 1];
  a[0] = last;
}

const left = [...arr];
const right = [...arr];
rotateLeft(left);
rotateRight(right);

console.log("Left =", left);
console.log("Right =", right);
💡 Hints

Stuck? Open one hint at a time

Hint 1

For a left rotation, save arr[0], shift everything left, then put the saved value at the end.

Hint 2

For a right rotation, save the last element, shift everything right, then put the saved value at index 0.

📌 Noted Points

Remember these

  • ⚡ Save first: Store the element that will be overwritten before you start shifting.
  • 💡 Copy the array: Each rotation function modifies its argument in O(n) time and O(1) auxiliary space. This demonstration makes two copies, so the complete example uses O(n) extra space. The early return protects empty and single-element arrays.
