const s = "leetcode";

const count = new Map();

for (const ch of s) count.set(ch, (count.get(ch) || 0) + 1);

let answer = "No non-repeating character";
for (const ch of s) {
  if (count.get(ch) === 1) {
    answer = ch;
    break;
  }
}

console.log(answer);

📌 Noted Points

Remember these

  • ⚡ Key pattern: Map. Two passes: count first, then find the first count of 1.
  • 💡 Second pass on the string: Loop over the string, not the Map, so you keep the original order.
  • ⚠️ No answer: A string like aabb has no unique character, so handle that case.
