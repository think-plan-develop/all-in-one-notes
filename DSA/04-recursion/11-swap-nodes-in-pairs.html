swap-nodes-in-pairs.js
class ListNode {
  constructor(val, next = null) {
    this.val = val;
    this.next = next;
  }
}

const build = (a) => a.reduceRight((next, val) => new ListNode(val, next), null);
const toArray = (node) => {
  const out = [];
  for (; node; node = node.next) out.push(node.val);
  return out;
};

const head = build([1, 2, 3, 4]);

function swapPairs(node) {
  if (!node || !node.next) return node;

  const second = node.next;
  node.next = swapPairs(second.next);
  second.next = node;

  return second;
}

console.log(toArray(swapPairs(head)));
💡 Hints

Stuck? Open one hint at a time

Hint 1

Base case: fewer than two nodes means nothing to swap.

Hint 2

Let second = node.next. Link node.next to the swapped rest of the list, then link second.next back to node.

📌 Noted Points

Remember these

  • ⚡ New head: Each call returns second, which becomes the new front of that pair.
  • 💡 Odd length: The last node has no partner and stays where it is.
  • ⚠️ Do not swap values: Interviewers expect the links to change, not the stored values.
