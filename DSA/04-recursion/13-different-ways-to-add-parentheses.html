const expression = "2-1-1";

const memo = new Map();

function compute(expr) {
  if (memo.has(expr)) return memo.get(expr);

  const results = [];
  for (let i = 0; i < expr.length; i++) {
    const ch = expr[i];
    if (ch === "+" || ch === "-" || ch === "*") {
      const left = compute(expr.slice(0, i));
      const right = compute(expr.slice(i + 1));
      for (const a of left) {
        for (const b of right) {
          if (ch === "+") results.push(a + b);
          else if (ch === "-") results.push(a - b);
          else results.push(a * b);
        }
      }
    }
  }

  if (results.length === 0) results.push(Number(expr));
  memo.set(expr, results);
  return results;
}

console.log(compute(expression));

Hint 1

Split at every operator. Solve the left part and the right part with recursion.

Hint 2

Combine every left result with every right result using that operator.

Hint 3

If the expression has no operator, it is just a number. Use a Map to save results of repeated sub-expressions.

📌 Noted Points

Remember these

  • ⚡ Base case: A string without an operator is a number, so return [Number(expr)].
  • 💡 Example: 2-1-1 can be (2-1)-1 = 0 or 2-(1-1) = 2.
  • ⚠️ Memoization: Sub-expressions repeat across different splits. Caching them avoids recomputing.
  • ✅ Order: Results can appear in any order.
