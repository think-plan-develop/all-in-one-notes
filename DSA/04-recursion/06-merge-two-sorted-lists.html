merge-two-sorted-lists.js
class ListNode {
  constructor(val, next = null) {
    this.val = val;
    this.next = next;
  }
}

const build = (a) => a.reduceRight((next, val) => new ListNode(val, next), null);
const toArray = (node) => {
  const out = [];
  for (; node; node = node.next) out.push(node.val);
  return out;
};

const list1 = build([1, 2, 4]);
const list2 = build([1, 3, 4]);

function merge(a, b) {
  if (!a) return b;
  if (!b) return a;

  if (a.val <= b.val) {
    a.next = merge(a.next, b);
    return a;
  }
  b.next = merge(a, b.next);
  return b;
}

console.log(toArray(merge(list1, list2)));
💡 Hints

Stuck? Open one hint at a time

Hint 1

If one list is empty, return the other list.

Hint 2

Pick the node with the smaller value. Its next becomes the merge of the rest.

📌 Noted Points

Remember these

  • ⚡ Base cases: When either list runs out, the remaining list is already sorted, so return it.
  • 💡 No new nodes: The existing nodes are relinked, so no extra list is created.
  • ⚠️ Cost: O(n + m) time and O(n + m) stack space.
