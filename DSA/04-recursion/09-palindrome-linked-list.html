palindrome-linked-list.js
class ListNode {
  constructor(val, next = null) {
    this.val = val;
    this.next = next;
  }
}

const build = (a) => a.reduceRight((next, val) => new ListNode(val, next), null);
const toArray = (node) => {
  const out = [];
  for (; node; node = node.next) out.push(node.val);
  return out;
};

const head = build([1, 2, 2, 1]);

let front = head;

function check(node) {
  if (!node) return true;

  if (!check(node.next)) return false;
  if (node.val !== front.val) return false;

  front = front.next;
  return true;
}

console.log(check(head));
💡 Hints

Stuck? Open one hint at a time

Hint 1

Keep a front pointer starting at the head. Recursion itself acts as the pointer moving from the back.

Hint 2

Recurse to the end first. On the way back, compare node.val with front.val, then move front forward.

📌 Noted Points

Remember these

  • ⚡ Why it works: The returning calls visit nodes from the back, while front moves from the start.
  • 💡 Space: O(n) because of the call stack. Reversing the second half uses O(1) space.
  • ⚠️ Early exit: Return false as soon as one pair does not match.
