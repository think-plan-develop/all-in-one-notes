recursive-binary-search.js
const arr = [1, 3, 5, 7, 9, 11];
const target = 7;

function search(a, target, low = 0, high = a.length - 1) {
  if (low > high) return -1;

  const mid = Math.floor((low + high) / 2);
  if (a[mid] === target) return mid;

  if (a[mid] < target) return search(a, target, mid + 1, high);
  return search(a, target, low, mid - 1);
}

console.log(search(arr, target));
💡 Hints

Stuck? Open one hint at a time

Hint 1

Look at the middle element. If it equals the target, return its index.

Hint 2

If the middle is smaller than the target, search the right half. Otherwise search the left half.

Hint 3

If low > high, the target is not in the array.

📌 Noted Points

Remember these

  • ⚡ Sorted only: Binary search works only on a sorted array.
  • 💡 Each call halves the range: Time is O(log n) and stack depth is also O(log n).
  • ⚠️ Mid calculation: Math.floor((low + high) / 2) is fine in JavaScript. In other languages use low + (high - low) / 2 to avoid overflow.
