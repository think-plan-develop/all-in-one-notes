const n = 4;

let depth = 0;
let maxDepth = 0;

function factorial(n) {
  depth++;
  maxDepth = Math.max(maxDepth, depth);
  console.log("  ".repeat(depth - 1) + "enter factorial(" + n + ")");

  const result = n <= 1 ? 1 : n * factorial(n - 1);

  console.log("  ".repeat(depth - 1) + "return " + result);
  depth--;
  return result;
}

console.log("Result =", factorial(n));
console.log("Max depth =", maxDepth);
💡 Hints

Stuck? Open one hint at a time

Hint 1

Keep a depth counter. Add 1 when a function starts and subtract 1 just before it returns.

Hint 2

Indent the logs with " ".repeat(depth - 1) so the stack growing and unwinding is easy to see.

📌 Noted Points

Remember these

  • ⚡ Stack frame: Every call gets its own frame holding its parameters, local variables and return address.
  • 💡 LIFO: The last function called is the first one to return, so the stack grows on the way down and unwinds on the way back.
  • ⚠️ Stack overflow: Too many frames cause RangeError: Maximum call stack size exceeded. Missing base cases are the usual cause.
  • ✅ Space: Recursion that goes n levels deep uses O(n) stack space.
