const arr = [8, 3, 15, 6, 10, 2];

function findMax(a, low, high) {
  if (low === high) return a[low];

  const mid = Math.floor((low + high) / 2);
  const left = findMax(a, low, mid);
  const right = findMax(a, mid + 1, high);

  return Math.max(left, right);
}

console.log(findMax(arr, 0, arr.length - 1));

Hint 1

Divide: split the range at mid.

Hint 2

Conquer: find the maximum of the left half and of the right half, using recursion.

Hint 3

Combine: the answer is Math.max(left, right). A range with one element is the base case.

📌 Noted Points

Remember these

  • ⚡ Three steps: Divide the problem, conquer the pieces recursively, combine their results.
  • 💡 Classic examples: Merge sort, quick sort, binary search and fast exponentiation.
  • ⚠️ Complexity: Described by recurrences like T(n) = 2T(n/2) + O(1), which gives O(n) here and O(n log n) for merge sort.
  • ✅ Independent parts: The subproblems must not overlap. If they overlap, use memoization instead.
