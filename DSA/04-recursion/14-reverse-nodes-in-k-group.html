class ListNode {
  constructor(val, next = null) {
    this.val = val;
    this.next = next;
  }
}

const build = (a) => a.reduceRight((next, val) => new ListNode(val, next), null);
const toArray = (node) => {
  const out = [];
  for (; node; node = node.next) out.push(node.val);
  return out;
};

const head = build([1, 2, 3, 4, 5]);
const k = 2;

function reverseKGroup(head, k) {
  let node = head;
  let count = 0;
  while (node && count < k) {
    node = node.next;
    count++;
  }
  if (count < k) return head;

  let prev = reverseKGroup(node, k);
  let curr = head;
  for (let i = 0; i < k; i++) {
    const next = curr.next;
    curr.next = prev;
    prev = curr;
    curr = next;
  }
  return prev;
}

console.log(toArray(reverseKGroup(head, k)));
💡 Hints

Stuck? Open one hint at a time

Hint 1

First check that k nodes exist. If fewer remain, return head unchanged.

Hint 2

Recursively solve the list that starts after those k nodes. Its result is where this group must connect.

Hint 3

Reverse the k nodes, making the last one point to that recursive result.

📌 Noted Points

Remember these

  • ⚡ Check first: Count k nodes before reversing, so a short last group is left alone.
  • 💡 Start prev as the rest: Setting prev to the recursive result links each reversed group to the next one.
  • ⚠️ Cost: O(n) time and O(n / k) stack depth.
