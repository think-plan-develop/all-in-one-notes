add-two-numbers.js
class ListNode {
  constructor(val, next = null) {
    this.val = val;
    this.next = next;
  }
}

const build = (a) => a.reduceRight((next, val) => new ListNode(val, next), null);
const toArray = (node) => {
  const out = [];
  for (; node; node = node.next) out.push(node.val);
  return out;
};

const l1 = build([2, 4, 3]);
const l2 = build([5, 6, 4]);

function add(a, b, carry = 0) {
  if (!a && !b && carry === 0) return null;

  const sum = (a ? a.val : 0) + (b ? b.val : 0) + carry;
  const node = new ListNode(sum % 10);
  node.next = add(a ? a.next : null, b ? b.next : null, Math.floor(sum / 10));

  return node;
}

console.log(toArray(add(l1, l2)));
💡 Hints

Stuck? Open one hint at a time

Hint 1

Each step adds the two digits and the carry. The new digit is sum % 10.

Hint 2

The new carry is Math.floor(sum / 10). Pass it into the next recursive call.

Hint 3

Stop when both lists are empty and there is no carry left.

📌 Noted Points

Remember these

  • ⚡ Final carry: [9, 9] + [1] gives [0, 0, 1]. The last carry needs its own node.
  • 💡 Different lengths: Treat a missing node as 0.
  • ⚠️ Example: 342 + 465 = 807, stored in reverse as [7, 0, 8].
