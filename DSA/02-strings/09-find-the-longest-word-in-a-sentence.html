function longestWord(sentence) {
  const words = sentence.match(/[A-Za-z]+/g) || [];
  let longest = "";
  for (const word of words) {
    if (word.length > longest.length) longest = word;
  }
  return longest;
}

const sentence = "I enjoy JavaScript programming.";
console.log(longestWord(sentence));

📌 Noted Points

Remember these

  • ⚡ Explicit word rule: Unlike question 5, this question uses letter-runs: "well-known" contains "well" and "known"; digits are ignored.
  • 💡 Tie example: "cat dog" returns "cat", because equal lengths do not replace the earlier answer.
  • ⚠️ Complexity: O(n) time and O(n) extra space for the extracted words.
  • ⚡ Character assumption: Use ASCII input for these exercises. JavaScript indexing works in UTF-16 code units; emoji and combined Unicode characters require a separate approach.
