function firstRepeatedCharacter(str) {
  const seen = new Set();
  for (const ch of str) {
    if (seen.has(ch)) return ch;
    seen.add(ch);
  }
  return null;
}

const str = "abccba";
console.log(firstRepeatedCharacter(str));

📌 Noted Points

Remember these

  • ⚡ Order matters: In "abccba", c repeats first during the scan, although a is the leftmost character that eventually repeats.
  • 💡 Case-sensitive: A and a are different. Spaces and punctuation also participate.
  • ⚠️ Complexity: Expected O(n) time and O(u) extra space with a hash set.
  • ⚡ Character assumption: Use ASCII input for these exercises. JavaScript indexing works in UTF-16 code units; emoji and combined Unicode characters require a separate approach.
