function isSubsequence(small, large) {
  let i = 0;
  for (let j = 0; j < large.length && i < small.length; j++) {
    if (small[i] === large[j]) i++;
  }
  return i === small.length;
}

const small = "abc";
const large = "ahbgdc";
console.log(isSubsequence(small, large));

📌 Noted Points

Remember these

  • ⚡ Subsequence vs substring: "abc" is a subsequence of "ahbgdc", but it is not a contiguous substring.
  • 💡 Order matters: "axc" is not a subsequence of "ahbgdc". Matching letters must occur in the required order.
  • ⚠️ Complexity: O(m) time and O(1) extra space, where m is large.length. The scan can stop once all characters of small match.
  • ⚡ Character assumption: Use ASCII input for these exercises. JavaScript indexing works in UTF-16 code units; emoji and combined Unicode characters require a separate approach.
