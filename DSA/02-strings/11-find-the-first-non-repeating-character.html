function firstUniqueCharacter(str) {
  const freq = new Map();
  for (const ch of str) freq.set(ch, (freq.get(ch) || 0) + 1);
  for (const ch of str) {
    if (freq.get(ch) === 1) return ch;
  }
  return null;
}

const str = "swiss";
console.log(firstUniqueCharacter(str));

📌 Noted Points

Remember these

  • ⚡ Two passes: The first pass establishes frequencies; the second preserves the original order.
  • 💡 No answer: "aabb" and "" return null. "aab" returns "b".
  • ⚠️ Complexity: Expected O(n) time and O(u) extra space with a hash map; u is the number of distinct characters.
  • ⚡ Character assumption: Use ASCII input for these exercises. JavaScript indexing works in UTF-16 code units; emoji and combined Unicode characters require a separate approach.
