function groupAnagrams(words) {
  const groups = new Map();
  for (const word of words) {
    const key = word.split("").sort().join("");
    if (!groups.has(key)) groups.set(key, []);
    groups.get(key).push(word);
  }
  return [...groups.values()];
}

const words = ["eat", "tea", "tan", "ate", "nat", "bat"];
console.log(groupAnagrams(words));

📌 Noted Points

Remember these

  • ⚡ Keep original words: Sort only the characters used to make the key. The returned words keep their original spelling.
  • 💡 Duplicates and empty words: ["", "", "a"] → [["", ""], ["a"]]. Repeated inputs are not removed.
  • ⚠️ Complexity: For N words of maximum length k, typical comparison sorting gives O(N × k log k) time when k ≥ 2; include O(N) overhead for empty/short words. O(N × (k + 1)) space is an upper bound for keys, groups and temporary characters.
  • ⚡ Character assumption: Use ASCII input for these exercises. JavaScript indexing works in UTF-16 code units; emoji and combined Unicode characters require a separate approach.
