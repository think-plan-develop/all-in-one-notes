function countCharacterTypes(str) {
  const counts = { letters: 0, digits: 0, spaces: 0, special: 0 };
  for (const ch of str) {
    if (/[A-Za-z]/.test(ch)) counts.letters++;
    else if (/[0-9]/.test(ch)) counts.digits++;
    else if (/\s/.test(ch)) counts.spaces++;
    else counts.special++;
  }
  return counts;
}

const str = "Hi 123!";
console.log(countCharacterTypes(str));

📌 Noted Points

Remember these

  • ⚡ Whitespace rule: Here spaces means whitespace. If an interviewer means only literal spaces, use ch === " " instead.
  • 💡 Consistency check: The four counts must add up to str.length under the ASCII assumption.
  • ⚠️ Complexity: O(n) time and O(1) extra space.
  • ⚡ Character assumption: Use ASCII input for these exercises. JavaScript indexing works in UTF-16 code units; emoji and combined Unicode characters require a separate approach.
