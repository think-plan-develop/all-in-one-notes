function reverseEachWord(sentence) {
  return sentence.replace(/\S+/g, (word) => {
    const chars = [];
    for (let i = word.length - 1; i >= 0; i--) {
      chars.push(word[i]);
    }
    return chars.join("");
  });
}

const sentence = "Hello  world!";
console.log(reverseEachWord(sentence));

📌 Noted Points

Remember these

  • ⚡ Keep spacing: Unlike question 7, this question preserves repeated spaces, tabs and newlines exactly.
  • 💡 Punctuation: "world!" becomes "!dlrow" because punctuation is part of the token.
  • ⚠️ Complexity: O(n) time and O(n) overall extra space including output and temporary character arrays.
  • ⚡ Character assumption: Use ASCII input for these exercises. JavaScript indexing works in UTF-16 code units; emoji and combined Unicode characters require a separate approach.
