function countWords(sentence) {
  let count = 0;
  let inWord = false;
  for (const ch of sentence) {
    if (/\s/.test(ch)) {
      inWord = false;
    } else if (!inWord) {
      count++;
      inWord = true;
    }
  }
  return count;
}

const sentence = "  Learn   JavaScript today!  ";
console.log(countWords(sentence));

Hint 2

Track whether you are already inside a word; increment only when entering one.

📌 Noted Points

Remember these

  • ⚡ Definition matters: "hello,world" counts as one token; "hello, world" counts as two. A punctuation-only token also counts under this rule.
  • 💡 Avoid empty tokens: Splitting on one literal space can create empty entries for repeated spaces. This loop avoids that issue.
  • ⚠️ Complexity: O(n) time and O(1) extra space.
  • ⚡ Character assumption: Use ASCII input for these exercises. JavaScript indexing works in UTF-16 code units; emoji and combined Unicode characters require a separate approach.
