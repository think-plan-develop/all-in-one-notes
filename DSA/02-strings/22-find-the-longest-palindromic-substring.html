function longestPalindromicSubstring(str) {
  let bestStart = 0;
  let bestLength = 0;
  function expand(left, right) {
    while (left >= 0 && right < str.length && str[left] === str[right]) {
      const length = right - left + 1;
      if (length > bestLength) {
        bestStart = left;
        bestLength = length;
      }
      left--;
      right++;
    }
  }
  for (let center = 0; center < str.length; center++) {
    expand(center, center);
    expand(center, center + 1);
  }
  return str.slice(bestStart, bestStart + bestLength);
}

const str = "babad";
console.log(longestPalindromicSubstring(str));
💡 Hints

Stuck? Open one hint at a time

Hint 1

Every palindrome has a center: one character for odd lengths, or a gap for even lengths.

Hint 2

Expand outwards while the characters match. Try both kinds of center at every position and remember the best range.

📌 Noted Points

Remember these

  • ⚡ Both centers: "racecar" has an odd center; "abba" has an even center. Checking only odd centers misses even-length answers.
  • 💡 Tie rule: Centers are visited from left to right. Updating only for a strictly longer result keeps the leftmost answer: "babad" → "bab".
  • ⚠️ Complexity: O(n²) worst-case time and O(1) working space, excluding the returned substring. Slice once at the end, not on every expansion.
  • ⚡ Character assumption: Use ASCII input for these exercises. JavaScript indexing works in UTF-16 code units; emoji and combined Unicode characters require a separate approach.
