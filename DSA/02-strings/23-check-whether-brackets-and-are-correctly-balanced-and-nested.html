function isValidBrackets(str) {
  const pairs = { ")": "(", "}": "{", "]": "[" };
  const stack = [];
  for (const ch of str) {
    if (ch === "(" || ch === "{" || ch === "[") {
      stack.push(ch);
    } else if (ch === ")" || ch === "}" || ch === "]") {
      if (stack.pop() !== pairs[ch]) return false;
    } else {
      return false;
    }
  }
  return stack.length === 0;
}

const str = "{[()]}";
console.log(isValidBrackets(str));

📌 Noted Points

Remember these

  • ⚡ Last opened, first closed: A stack checks nesting, not just counts. "([)]" has equal counts but is invalid.
  • 💡 Return true at the end: Do not return true inside the loop when the stack becomes empty: a later bracket may still be invalid. "() (" is invalid under the stated rules.
  • ⚠️ Complexity: O(n) time and O(n) worst-case extra space, for example when all characters are opening brackets.
  • ⚡ Character assumption: Use ASCII input for these exercises. JavaScript indexing works in UTF-16 code units; emoji and combined Unicode characters require a separate approach.
