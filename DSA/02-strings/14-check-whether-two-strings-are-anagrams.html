function areAnagrams(a, b) {
  if (a.length !== b.length) return false;
  const freq = new Map();
  for (const ch of a) freq.set(ch, (freq.get(ch) || 0) + 1);
  for (const ch of b) {
    const count = freq.get(ch) || 0;
    if (count === 0) return false;
    freq.set(ch, count - 1);
  }
  return true;
}

const a = "listen";
const b = "silent";
console.log(areAnagrams(a, b));

📌 Noted Points

Remember these

  • ⚡ Frequency matters: A Set alone is insufficient: "aab" and "abb" have the same distinct letters but different counts.
  • 💡 Normalization: "Listen" and "silent" are false here. Normalize case or remove spaces only if the question explicitly requests it.
  • ⚠️ Complexity: Expected O(n + m) time and O(u) extra space; n and m are the input lengths.
  • ⚡ Character assumption: Use ASCII input for these exercises. JavaScript indexing works in UTF-16 code units; emoji and combined Unicode characters require a separate approach.
