function isPalindrome(str) {
  let left = 0;
  let right = str.length - 1;
  while (left < right) {
    if (str[left] !== str[right]) return false;
    left++;
    right--;
  }
  return true;
}

const str = "madam";
console.log(isPalindrome(str));

📌 Noted Points

Remember these

  • ⚡ Exact comparison: "Madam" is false here because M and m differ. Question 16 covers ignoring case and punctuation.
  • 💡 Complexity: O(n) time and O(1) extra space; no reversed copy is needed.
  • ⚠️ Check: "" → true; "abba" → true; "abca" → false.
  • ⚡ Character assumption: Use ASCII input for these exercises. JavaScript indexing works in UTF-16 code units; emoji and combined Unicode characters require a separate approach.
