function countVowelsConsonants(str) {
  const vowels = new Set(["a", "e", "i", "o", "u"]);
  let vowelCount = 0;
  let consonantCount = 0;
  for (const ch of str) {
    const lower = ch.toLowerCase();
    if (lower >= "a" && lower <= "z") {
      if (vowels.has(lower)) vowelCount++;
      else consonantCount++;
    }
  }
  return { vowels: vowelCount, consonants: consonantCount };
}

const str = "Hello World!";
console.log(countVowelsConsonants(str));

📌 Noted Points

Remember these

  • ⚡ Ignore non-letters: Do not count every non-vowel as a consonant: "1", "!" and spaces belong to neither group.
  • 💡 Complexity: O(n) time and O(1) extra space; the vowel set has a fixed size of five.
  • ⚠️ Check: "AEIOU" → {vowels: 5, consonants: 0}; "123!" → both 0.
  • ⚡ Character assumption: Use ASCII input for these exercises. JavaScript indexing works in UTF-16 code units; emoji and combined Unicode characters require a separate approach.
