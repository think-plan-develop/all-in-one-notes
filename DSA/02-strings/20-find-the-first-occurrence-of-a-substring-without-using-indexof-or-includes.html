function findSubstring(text, pattern) {
  if (pattern.length === 0) return 0;
  if (pattern.length > text.length) return -1;
  for (let start = 0; start <= text.length - pattern.length; start++) {
    let j = 0;
    while (j < pattern.length && text[start + j] === pattern[j]) {
      j++;
    }
    if (j === pattern.length) return start;
  }
  return -1;
}

const text = "hello";
const pattern = "ll";
console.log(findSubstring(text, pattern));

Hint 1

Try every starting index that leaves enough characters for the pattern.

Hint 2

Compare pattern characters in order. Return the current start only if every character matches.

📌 Noted Points

Remember these

  • ⚡ Boundary: Use start <= text.length - pattern.length so the final possible starting index is checked.
  • 💡 Contiguous match: Unlike a subsequence, a substring must match adjacent characters.
  • ⚠️ Complexity: For non-empty pattern of length m ≤ n, worst-case O((n - m + 1) × m) time and O(1) extra space. KMP is a later O(n + m) algorithm, not required for this baseline.
  • ⚡ Character assumption: Use ASCII input for these exercises. JavaScript indexing works in UTF-16 code units; emoji and combined Unicode characters require a separate approach.
