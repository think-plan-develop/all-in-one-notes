function canBePalindrome(str) {
  function isRangePalindrome(left, right) {
    while (left < right) {
      if (str[left] !== str[right]) return false;
      left++;
      right--;
    }
    return true;
  }
  let left = 0;
  let right = str.length - 1;
  while (left < right) {
    if (str[left] !== str[right]) {
      return isRangePalindrome(left + 1, right) ||
             isRangePalindrome(left, right - 1);
    }
    left++;
    right--;
  }
  return true;
}

const str = "abca";
console.log(canBePalindrome(str));

Hint 1

Move inward while the outer characters match.

At the first mismatch, try skipping the left character OR the right character. The remaining range must be a palindrome with no more deletions.

📌 Noted Points

Remember these

  • ⚡ At most one: An already-valid palindrome needs zero deletions and must return true.
  • 💡 Try both sides: For "abca", deleting b or c works. Always deleting the left mismatched character can miss valid answers.
  • ⚠️ Complexity: O(n) time and O(1) extra space. At most two additional linear scans are performed, without slicing.
  • ⚡ Character assumption: Use ASCII input for these exercises. JavaScript indexing works in UTF-16 code units; emoji and combined Unicode characters require a separate approach.
