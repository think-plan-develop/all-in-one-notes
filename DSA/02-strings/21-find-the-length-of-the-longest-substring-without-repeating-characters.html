function longestUniqueSubstringLength(str) {
  const lastSeen = new Map();
  let left = 0;
  let best = 0;
  for (let right = 0; right < str.length; right++) {
    const ch = str[right];
    if (lastSeen.has(ch)) {
      left = Math.max(left, lastSeen.get(ch) + 1);
    }
    lastSeen.set(ch, right);
    best = Math.max(best, right - left + 1);
  }
  return best;
}

const str = "abcabcbb";
console.log(longestUniqueSubstringLength(str));

Hint 1

Maintain a window with no repeated characters and remember the last index of each character.

Hint 2

When a character repeats inside the window, move the left edge just after its previous occurrence. Never move left backwards.

📌 Noted Points

Remember these

  • ⚡ Why Math.max: In "abba", the final a was last seen before the current window. Moving left backwards would incorrectly allow a repeated b.
  • 💡 Substring, not subsequence: "pwwkew" gives 3 ("wke" or "kew"); "pwke" is not contiguous.
  • ⚠️ Complexity: Expected O(n) time and O(u) extra space with a hash map. Each right index is processed once.
  • ⚡ Character assumption: Use ASCII input for these exercises. JavaScript indexing works in UTF-16 code units; emoji and combined Unicode characters require a separate approach.
