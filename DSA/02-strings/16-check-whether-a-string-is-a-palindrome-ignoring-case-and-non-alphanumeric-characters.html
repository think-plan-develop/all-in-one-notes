function isNormalizedPalindrome(str) {
  const cleaned = str.toLowerCase().replace(/[^a-z0-9]/g, "");
  let left = 0;
  let right = cleaned.length - 1;
  while (left < right) {
    if (cleaned[left] !== cleaned[right]) return false;
    left++;
    right--;
  }
  return true;
}

const str = "A man, a plan, a canal: Panama!";
console.log(isNormalizedPalindrome(str));

📌 Noted Points

Remember these

  • ⚡ Different from question 2: This question deliberately ignores case and punctuation; the earlier palindrome question compares every character exactly.
  • 💡 Empty after cleaning: "!!!" returns true. "0P" returns false because 0 and p differ.
  • ⚠️ Complexity: O(n) time and O(n) extra space for the normalized copy. A later optimization can skip non-alphanumeric characters directly with pointers.
  • ⚡ Character assumption: Use ASCII input for these exercises. JavaScript indexing works in UTF-16 code units; emoji and combined Unicode characters require a separate approach.
