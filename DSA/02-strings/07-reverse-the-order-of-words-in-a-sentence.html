function reverseWordOrder(sentence) {
  const words = sentence.trim().split(/\s+/);
  const result = [];
  for (let i = words.length - 1; i >= 0; i--) {
    result.push(words[i]);
  }
  return result.join(" ");
}

const sentence = "  I   love JavaScript  ";
console.log(reverseWordOrder(sentence));

📌 Noted Points

Remember these

  • ⚡ Word order only: "hello world" becomes "world hello", not "dlrow olleh".
  • 💡 Empty input: Trimming an empty or whitespace-only sentence produces "". The split result [""] still joins back to "" correctly.
  • ⚠️ Complexity: O(n) time and O(n) extra space, where n is the sentence length.
  • ⚡ Character assumption: Use ASCII input for these exercises. JavaScript indexing works in UTF-16 code units; emoji and combined Unicode characters require a separate approach.
