FILE BREAKDOWN
01

AppError.ts & subclasses

🔴
AppError.ts
src/errors/AppError.ts
DEFINED HERE
What it does: The single base error class for the whole app. Every custom error extends it. It adds statusCode, isOperational (so we know it's an expected error, not a bug), and an optional errors payload for field-level detail. Nothing throws raw new Error() — everything uses this.
IMPORTED BY → auth.ts authorise.ts user.service.ts errorHandler.ts user.routes.ts (via middleware) handleMongooseError.ts 
// ═══════════════════════════════════════════════════════════
// src/errors/AppError.ts
// DEFINED HERE — imported by every other layer
// ═══════════════════════════════════════════════════════════

// ── Base class ─────────────────────────────────────────────
export class AppError extends Error {
  public readonly statusCode:    number;
  public readonly isOperational: boolean;   // true = known/expected error
  public readonly errors?:       unknown;    // field-level detail (optional)

  constructor(
    message:    string,
    statusCode: number = 500,
    errors?:    unknown
  ) {
    super(message);
    this.name          = this.constructor.name;  // "NotFoundError" etc.
    this.statusCode    = statusCode;
    this.isOperational = true;
    this.errors        = errors;
    Error.captureStackTrace(this, this.constructor);
  }
}

// ── Subclasses — convenience wrappers ─────────────────────
// Import + throw these directly. No magic number status codes.

export class NotFoundError extends AppError {
  constructor(resource = "Resource") {
    super(`${resource} not found`, 404);
  }
}

export class ValidationError extends AppError {
  constructor(errors: unknown) {
    super("Validation failed", 400, errors);
  }
}

export class UnauthorizedError extends AppError {
  constructor(message = "Unauthorized") {
    super(message, 401);
  }
}

export class ForbiddenError extends AppError {
  constructor(message = "Forbidden") {
    super(message, 403);
  }
}

export class ConflictError extends AppError {
  constructor(field: string) {
    super(`${field} already exists`, 409);
  }
}
auth.ts
throw new UnauthorizedError("No token provided") when the Authorization header is missing, and throw new UnauthorizedError("Token expired") when JWT verification fails. Passed to next(err) so errorHandler responds with 401.
authorise.ts
throw new ForbiddenError(`Role '${role}' not allowed`) when the authenticated user's role is not in the allowed list. Responds 403.
user.service.ts
throw new NotFoundError("User") when findById returns null. throw new ConflictError("email") when email already exists. throw new ForbiddenError("...") on ownership checks. All respond with the matching 4xx status.
handleMongooseError.ts
Does NOT import AppError. It returns a plain object { statusCode, body } that errorHandler sends directly. Mongoose errors bypass the class hierarchy.
errorHandler.ts
Checks err instanceof AppError to detect known errors and reads err.statusCode, err.message, err.errors to build the response. This is the only consumer that reads AppError — everyone else throws it.
GLOBAL HANDLER
02

errorHandler.ts

🟣
errorHandler.ts
src/middleware/errorHandler.ts
DEFINED HERE
What it does: The single Express error middleware registered last in app.ts. Every unhandled error in the app flows here via next(err). It identifies the error type and formats a consistent JSON response. It never throws — it only responds.
REGISTERED IN → app.ts (must be LAST middleware) RECEIVES ERRORS FROM → asyncHandler (via next) auth.ts (via next) authorise.ts (via next) validate.ts (via next) 
// ═══════════════════════════════════════════════════════════
// src/middleware/errorHandler.ts
// REGISTERED LAST in app.ts — catches every next(err)
// ═══════════════════════════════════════════════════════════

import { Request, Response, NextFunction } from "express";
import { ZodError }              from "zod";
import { AppError }             from "../errors/AppError";
import { handleMongooseError }   from "./handleMongooseError";

const isDev = process.env.NODE_ENV === "development";

export const errorHandler = (
  err:   unknown,
  _req:  Request,
  res:   Response,
  _next: NextFunction   // 4-param signature required by Express
): void => {

  // ── Step 1: Zod validation error (from validate middleware) ─
  if (err instanceof ZodError) {
    res.status(400).json({
      success: false,
      message: "Validation failed",
      errors:  err.flatten().fieldErrors    // per-field error map
    });
    return;
  }

  // ── Step 2: Known AppError (our custom errors) ─────────────
  if (err instanceof AppError) {
    res.status(err.statusCode).json({
      success: false,
      message: err.message,
      ...(err.errors && { errors: err.errors }),
      ...(isDev      && { stack:  err.stack  })   // show stack in dev only
    });
    return;
  }

  // ── Step 3: Mongoose errors (CastError, ValidationError, 11000) ─
  const mongoResult = handleMongooseError(err);
  if (mongoResult) {
    res.status(mongoResult.statusCode).json(mongoResult.body);
    return;
  }

  // ── Step 4: Unknown / programmer bug → 500 ─────────────────
  console.error("[UNHANDLED]", err);
  res.status(500).json({
    success: false,
    message: isDev
      ? (err instanceof Error ? err.message : "Unknown error")
      : "Internal server error",   // never leak internals in prod
    ...(isDev && { stack: err instanceof Error ? err.stack : undefined })
  });
};

// ──────────────────────────────────────────────────────────
// src/app.ts  (where this is registered)
// ──────────────────────────────────────────────────────────
app.use("/users", userRouter);
app.use("/posts", postRouter);
app.use(notFound);       // 404 for unknown routes — AFTER all routes
app.use(errorHandler);  // ← MUST be last. Express identifies by 4 params.
app.ts
app.use(errorHandler) — registered as the very last middleware. Express recognises the 4-argument signature (err, req, res, next) and routes all next(err) calls here automatically.
asyncHandler.ts
.catch(next) inside asyncHandler forwards any thrown error from controllers or services directly to this handler.
auth.ts
return next(new UnauthorizedError(...)) — explicit forwarding from the middleware function.
validate.ts
return next(zodError) — Zod errors are passed to next and Step 1 of errorHandler catches them.
MONGOOSE NORMALISER
03

handleMongooseError.ts

🟡
handleMongooseError.ts
src/middleware/handleMongooseError.ts
DEFINED HERE
What it does: Mongoose throws its own error types that don't extend AppError. This helper detects them by class or error code and converts them to a plain { statusCode, body } object. Called only by errorHandler. Returns null when the error isn't a Mongoose error.
CALLED BY → errorHandler.ts (step 3 only) 
// ═══════════════════════════════════════════════════════════
// src/middleware/handleMongooseError.ts
// CALLED ONLY FROM errorHandler — Step 3
// ═══════════════════════════════════════════════════════════

import mongoose from "mongoose";

interface NormalisedError {
  statusCode: number;
  body: { success: false; message: string; errors?: unknown };
}

export const handleMongooseError = (err: unknown): NormalisedError | null => {

  // ── Duplicate key (unique index violation) ─────────────────
  // Thrown by: User.create({ email: "taken@mail.com" })
  // MongoDB error code 11000
  if (
    err instanceof Error &&
    "code" in err &&
    (err as any).code === 11000
  ) {
    const field = Object.keys((err as any).keyValue ?? {})[0] ?? "field";
    return {
      statusCode: 409,
      body: { success: false, message: `${field} already exists` }
    };
  }

  // ── Schema ValidationError ─────────────────────────────────
  // Thrown by: User.create({ age: 5 }) — violates min: 18
  // Thrown by: User.create({}) — missing required fields
  if (err instanceof mongoose.Error.ValidationError) {
    const errors = Object.fromEntries(
      Object.entries(err.errors).map(([key, val]) => [key, val.message])
    );
    return {
      statusCode: 400,
      body: { success: false, message: "Validation failed", errors }
    };
  }

  // ── CastError ──────────────────────────────────────────────
  // Thrown by: User.findById("not-a-valid-objectid")
  // Mongoose can't cast the string to ObjectId
  if (err instanceof mongoose.Error.CastError) {
    return {
      statusCode: 400,
      body: {
        success: false,
        message: `Invalid ${err.path}: "${err.value}" is not a valid ID`
      }
    };
  }

  return null;  // not a Mongoose error — let errorHandler fall to 500
};
errorHandler.ts
const mongoResult = handleMongooseError(err) — called in Step 3. If the return value is non-null, res.status(mongoResult.statusCode).json(mongoResult.body) is sent immediately and the function returns.
ASYNC WRAPPER
04

asyncHandler.ts

🟢
asyncHandler.ts
src/utils/asyncHandler.ts
DEFINED HERE
What it does: Wraps every async controller function. Without it, a rejected promise in a controller silently disappears (Express doesn't catch async errors). .catch(next) forwards the rejection to errorHandler automatically — no try/catch boilerplate needed in any controller.
IMPORTED BY → user.controller.ts post.controller.ts auth.controller.ts FORWARDS TO → errorHandler.ts 
// ═══════════════════════════════════════════════════════════
// src/utils/asyncHandler.ts
// IMPORTED BY every controller file
// ═══════════════════════════════════════════════════════════

import { Request, Response, NextFunction, RequestHandler } from "express";

type AsyncFn = (req: Request, res: Response, next: NextFunction)
  => Promise<void>;

export const asyncHandler =
  (fn: AsyncFn): RequestHandler =>
  (req, res, next) =>
    fn(req, res, next).catch(next);
//                    ^^^^^^^^^^^
//  This single .catch(next) replaces try/catch in EVERY controller.
//  "next" here is Express's next() — it routes to errorHandler.

// ── Without asyncHandler (bad) ─────────────────────────────
router.get("/:id", async (req, res, next) => {
  try {
    const user = await userService.getUserById(req.params.id);
    res.json(user);
  } catch (err) {
    next(err);  // have to remember this in every single handler!
  }
});

// ── With asyncHandler (good) ───────────────────────────────
router.get("/:id", asyncHandler(async (req, res) => {
  const user = await userService.getUserById(req.params.id);
  res.json(user);
  // service throws NotFoundError → .catch(next) → errorHandler → 404
}));
user.controller.ts
Every exported handler is wrapped: create: asyncHandler(async (req, res) => { ... }). Any throw inside is caught and forwarded to errorHandler.
post.controller.ts
Same pattern. asyncHandler is imported at the top and wraps every method — create, getAll, getById, update, remove.
AUTH MIDDLEWARE
05

auth.ts + authorise.ts

🔵
auth.ts & authorise.ts
src/middleware/auth.ts  ·  src/middleware/authorise.ts
DEFINED HERE
What it does: authenticate runs before every protected route. It reads the JWT from Authorization: Bearer <token>, verifies it, and attaches the payload to req.user. authorise then checks that req.user.role is in the allowed list. Both throw and call next(err) rather than responding directly.
USED IN → user.routes.ts post.routes.ts THROWS TO → errorHandler.ts 
// ═══════════════════════════════════════════════════════════
// src/middleware/auth.ts — throws UnauthorizedError (401)
// ═══════════════════════════════════════════════════════════

import jwt                        from "jsonwebtoken";
import { Request, Response, NextFunction } from "express";
import { UnauthorizedError }     from "../errors/AppError";  // ← imports AppError

export const authenticate = (req: Request, _res: Response, next: NextFunction) => {
  const token = req.headers.authorization?.split(" ")[1];

  // THROW POINT A — no token present at all
  if (!token) return next(new UnauthorizedError("No token provided"));

  try {
    const payload = jwt.verify(token, process.env.JWT_SECRET!);
    (req as any).user = payload;
    next();

  } catch (err) {
    // THROW POINT B — token expired
    if (err instanceof jwt.TokenExpiredError)
      return next(new UnauthorizedError("Token expired — please log in again"));

    // THROW POINT C — token tampered / invalid
    return next(new UnauthorizedError("Invalid token"));
  }
};

// ═══════════════════════════════════════════════════════════
// src/middleware/authorise.ts — throws ForbiddenError (403)
// ═══════════════════════════════════════════════════════════

import { ForbiddenError } from "../errors/AppError";  // ← imports AppError
import { UserRole }      from "../types";

export const authorise = (...roles: UserRole[]) =>
  (req: Request, _res: Response, next: NextFunction) => {
    const role = (req as any).user?.role;

    // THROW POINT D — user is logged in but wrong role
    if (!roles.includes(role))
      return next(new ForbiddenError(`Role '${role}' cannot access this resource`));

    next();
  };
user.routes.ts
router.get("/:id", authenticate, controller.getById) — authenticate runs first. If it calls next(err), Express skips the controller and goes straight to errorHandler. authorise("ADMIN") is chained after authenticate on admin-only routes.
post.routes.ts
Same pattern: router.delete("/:id", authenticate, authorise("ADMIN", "MANAGER"), controller.remove). Three middlewares chain — only the last one is the controller.
SERVICE LAYER
06

user.service.ts — primary throw layer

🩷
user.service.ts
src/services/user.service.ts
PRIMARY THROW LAYER
What it does: Contains all business logic. This is where the majority of throw new XxxError() calls live. The controller calls it, asyncHandler catches any throw, and it flows to errorHandler. The service never touches res or next — it only throws or returns data.
CALLED BY → user.controller.ts IMPORTS → AppError / subclasses THROWS REACH → errorHandler.ts (via asyncHandler) 
// ═══════════════════════════════════════════════════════════
// src/services/user.service.ts
// PRIMARY THROW LAYER — business logic lives here
// ═══════════════════════════════════════════════════════════

import { User }                           from "../models/user.model";
import { CreateUserDTO }                 from "../types";
import {
  NotFoundError,
  ConflictError,
  ForbiddenError,
  AppError
} from "../errors/AppError";   // ← every throw comes from here
import bcrypt from "bcryptjs";

export const userService = {

  createUser: async (dto: CreateUserDTO) => {
    // THROW POINT 1 — duplicate email (explicit pre-check)
    const existing = await User.findOne({ email: dto.email });
    if (existing) throw new ConflictError("email");     // → 409

    const hashed = await bcrypt.hash(dto.password, 12);
    return User.create({ ...dto, password: hashed });
    // if mongoose throws 11000 here anyway → handleMongooseError catches it
  },

  getUserById: async (id: string) => {
    // THROW POINT 2 — resource not found
    const user = await User.findById(id);
    if (!user) throw new NotFoundError("User");          // → 404
    return user;
    // bad id format → Mongoose CastError → handleMongooseError → 400
  },

  updateUser: async (id: string, dto: Partial<CreateUserDTO>, requesterId: string) => {
    const user = await User.findById(id);

    // THROW POINT 3 — not found
    if (!user) throw new NotFoundError("User");          // → 404

    // THROW POINT 4 — ownership check
    if (user._id.toString() !== requesterId)
      throw new ForbiddenError("You can only edit your own profile");  // → 403

    // THROW POINT 5 — business rule violation
    if (dto.role && dto.role !== user.role)
      throw new AppError("You cannot change your own role", 403);     // → 403

    return User.findByIdAndUpdate(id, { $set: dto }, { new: true, runValidators: true });
  },

  deleteUser: async (id: string) => {
    // THROW POINT 6 — delete target not found
    const deleted = await User.findByIdAndDelete(id);
    if (!deleted) throw new NotFoundError("User");      // → 404
  },

};
user.controller.ts
Every method calls a service function inside asyncHandler: e.g. const user = await userService.getUserById(req.params.id). If service throws, .catch(next) forwards to errorHandler. The controller never sees the error.
ROUTE WIRING
07

user.routes.ts — everything wired

🟠
user.routes.ts
src/routes/user.routes.ts
WIRES IT ALL
What it does: Connects every error-handling piece in the correct order per route. The route layer itself never throws — it just attaches the right chain of middleware. Reading this file shows the complete error path for any endpoint at a glance.
IMPORTS → validate (Zod MW) authenticate authorise userController MOUNTED IN → app.ts 
// ═══════════════════════════════════════════════════════════
// src/routes/user.routes.ts
// WIRES: validate → authenticate → authorise → controller
// ═══════════════════════════════════════════════════════════

import { Router }              from "express";
import { validate }            from "../middleware/validate";
import { authenticate }        from "../middleware/auth";
import { authorise }           from "../middleware/authorise";
import { userController }      from "../controllers/user.controller";
import { createUserSchema, updateUserSchema } from "../validators/user.validator";

const router = Router();

// ── POST /users ────────────────────────────────────────────
// Error path: Zod fails → validate throws ZodError → errorHandler → 400
// Error path: email taken → service throws ConflictError → errorHandler → 409
router.post("/",
  validate(createUserSchema),        // 1. Zod: bad body → ZodError → 400
  userController.create             // 2. service: duplicate → ConflictError → 409
);

// ── GET /users/:id ─────────────────────────────────────────
// Error path: no JWT → authenticate throws UnauthorizedError → 401
// Error path: bad ID  → Mongoose CastError → handleMongooseError → 400
// Error path: not found → service throws NotFoundError → 404
router.get("/:id",
  authenticate,                      // 1. JWT missing/bad → 401
  userController.getById            // 2. service: not found → 404
);

// ── PATCH /users/:id ──────────────────────────────────────
// Error path: no JWT     → 401
// Error path: wrong role → 403
// Error path: bad body   → 400 (Zod)
// Error path: not found  → 404
// Error path: not owner  → 403 (service ownership check)
router.patch("/:id",
  authenticate,                      // 1. JWT check → 401
  authorise("ADMIN", "MANAGER", "USER"), // 2. role check → 403
  validate(updateUserSchema),        // 3. Zod: bad body → 400
  userController.update             // 4. service: not found / not owner → 404/403
);

// ── DELETE /users/:id ─────────────────────────────────────
// Error path: no JWT     → 401
// Error path: wrong role → 403 (ADMIN only)
// Error path: not found  → 404
router.delete("/:id",
  authenticate,                      // 1. JWT check → 401
  authorise("ADMIN"),               // 2. ADMIN only → 403
  userController.remove             // 3. service: not found → 404
);

export { router as userRouter };
app.ts
app.use("/users", userRouter) — mounts the router. After all routers: app.use(notFound) then app.use(errorHandler). The order is critical — errorHandler must be last.
