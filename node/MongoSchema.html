👤 User Schema

import mongoose from "mongoose";

const userSchema = new mongoose.Schema({

  firstName: {
    type: String,
    required: true,
    trim: true,
    minlength: 2,
    maxlength: 50
  },

  lastName: {
    type: String,
    required: true
  },

  email: {
    type: String,
    required: true,
    unique: true,
    lowercase: true
  },

  age: {
    type: Number,
    min: 18,
    max: 100
  },

  isActive: {
    type: Boolean,
    default: true
  },

  birthDate: {
    type: Date
  },

  salary: {
    type: mongoose.Schema.Types.Decimal128
  },

  profileImage: {
    type: Buffer
  },

  role: {
    type: String,
    enum: ["ADMIN", "USER", "MANAGER"],
    default: "USER"
  },

  skills: [{
    type: String
  }],

  phoneNumbers: [{
    type: Number
  }],

  address: {
    city: String,
    state: String,
    country: String,
    zipCode: Number
  },

  socialLinks: {
    github: String,
    linkedin: String,
    website: String
  },

  settings: {
    type: Map,
    of: String
  },

  posts: [{
    type: mongoose.Schema.Types.ObjectId,
    ref: "Post"
  }]

}, {
  timestamps: true
});

export const User = mongoose.model("User", userSchema);

📝 Post Schema

import mongoose from "mongoose";

const commentSchema = new mongoose.Schema({
  message: String,
  commentedBy: {
    type: mongoose.Schema.Types.ObjectId,
    ref: "User"
  }
});

const postSchema = new mongoose.Schema({

  title: {
    type: String,
    required: true
  },

  description: {
    type: String,
    required: true
  },

  author: {
    type: mongoose.Schema.Types.ObjectId,
    ref: "User",
    required: true
  },

  tags: [{
    type: String
  }],

  likes: {
    type: Number,
    default: 0
  },

  comments: [commentSchema]

}, {
  timestamps: true
});

export const Post = mongoose.model("Post", postSchema);

Installation

npm install zod

User Validation

import { z } from "zod";

export const createUserSchema = z.object({
  firstName: z.string()
    .min(2)
    .max(50),

  lastName: z.string(),

  email: z.email(),

  age: z.number()
    .min(18)
    .max(100),

  role: z.enum([
    "ADMIN",
    "USER",
    "MANAGER"
  ]),

  skills: z.array(z.string())
});

🛡️ Validation Middleware

export const validate = (schema) => (req, res, next) => {
  const result = schema.safeParse(req.body);

  if (!result.success) {
    return res.status(400).json({
      success: false,
      errors: result.error
    });
  }

  next();
};

📤 POST Request (Create)

Purpose:

Create Resource

Endpoint:

POST /users

Request Example:

{
  "firstName":"Shivam",
  "lastName":"Singh",
  "email":"shivam@gmail.com",
  "age":25,
  "role":"USER",
  "skills":["Node","MongoDB"]
}

Flow:

Route ↓ Validation ↓ Controller ↓ Service ↓ Model ↓ MongoDB

📥 GET Request (Read)

Purpose:

Fetch Resource

Endpoints:

GET /users
GET /users/:id

Examples:

User.find();

User.findById(id);

User.find({ role: "USER" });

🔄 PUT Request (Replace)

Purpose:

Replace Entire Resource

Endpoint:

PUT /users/:id

Request Example:

{
  "firstName":"Updated",
  "lastName":"User",
  "email":"updated@gmail.com",
  "age":30
}

Example Code:

User.findByIdAndUpdate(
  id,
  req.body,
  {
    new: true,
    overwrite: true
  }
);

🔧 PATCH Request (Partial Update)

Purpose:

Update Selected Fields

Endpoint:

PATCH /users/:id

Request Example:

{
  "age":35
}

Example Code:

User.findByIdAndUpdate(
  id,
  {
    age: 35
  },
  {
    new: true
  }
);

🗑️ DELETE Request

Purpose:

Remove Resource

Endpoint:

Example Code:

User.findByIdAndDelete(id);

🔗 Population

Instead of storing just the ID reference:

{
  "author":"6851abc123"
}

Fetch the full related document:

Post.find()
  .populate("author");

Result:

{
  "author":{
    "name":"Shivam",
    "email":"shivam@gmail.com"
  }
}
