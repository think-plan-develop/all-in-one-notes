import mongoose, { Schema, InferSchemaType, Model } from "mongoose";

// ── User Schema ────────────────────────────────────────────
const userSchema = new Schema({

  // Identity
  firstName: { type: String,  required: true,  trim: true, minlength: 2, maxlength: 50 },
  lastName:  { type: String,  required: true,  trim: true },
  email:     { type: String,  required: true,  unique: true, lowercase: true, trim: true },

  // Auth & Access
  role: {
    type:    String,
    enum:    ["ADMIN", "USER", "MANAGER"] as const,
    default: "USER"
  },
  isActive: { type: Boolean, default: true },
  password: { type: String, required: true, select: false },  // hidden by default

  // Profile
  age:          { type: Number,                  min: 18, max: 100 },
  birthDate:    { type: Date  },
  salary:       { type: Schema.Types.Decimal128 },
  profileImage: { type: String },                               // URL or base64

  // Collections
  skills:       [{ type: String, trim: true }],
  phoneNumbers: [{ type: String }],                             // String keeps leading 0s

  // Embedded sub-documents (always fetched with user)
  address: {
    city:    { type: String },
    state:   { type: String },
    country: { type: String },
    zipCode: { type: String }
  },

  socialLinks: {
    github:   { type: String },
    linkedin: { type: String },
    website:  { type: String }
  },

  // Dynamic key-value settings map
  settings: { type: Map, of: String },

  // References (normalised — loaded via populate)
  posts: [{ type: Schema.Types.ObjectId, ref: "Post" }]

}, {
  timestamps: true                                              // adds createdAt, updatedAt
});

// ── Indexes ────────────────────────────────────────────────
userSchema.index({ email: 1 });
userSchema.index({ role: 1, isActive: 1 });

// ── Infer TypeScript type directly from schema ─────────────
export type IUser = InferSchemaType<typeof userSchema>;
// IUser is now fully typed — no separate interface needed

// ── Typed Model ────────────────────────────────────────────
export const User: Model<IUser> = mongoose.model<IUser>("User", userSchema);

// ── Comment Sub-Document Schema ────────────────────────────
const commentSchema = new Schema({
  message:     { type: String, required: true, trim: true },
  commentedBy: { type: Schema.Types.ObjectId, ref: "User", required: true }
}, {
  _id:        true,
  timestamps: true
});

export type IComment = InferSchemaType<typeof commentSchema>;

// ── Post Schema ────────────────────────────────────────────
const postSchema = new Schema({

  title:       { type: String, required: true, trim: true, maxlength: 200 },
  description: { type: String, required: true },
  slug:        { type: String, unique: true, lowercase: true },  // URL-friendly ID

  author: {
    type:     Schema.Types.ObjectId,
    ref:      "User",
    required: true
  },

  tags:   [{ type: String, lowercase: true, trim: true }],
  likes:  { type: Number, default: 0, min: 0 },
  views:  { type: Number, default: 0, min: 0 },

  isPublished: { type: Boolean, default: false },

  comments: [commentSchema]    // embedded — always loaded with post

}, {
  timestamps: true
});

// ── Indexes ────────────────────────────────────────────────
postSchema.index({ author: 1, createdAt: -1 });
postSchema.index({ tags: 1 });
postSchema.index({ slug: 1 }, { unique: true, sparse: true });

// ── Pre-save hook: auto-generate slug ─────────────────────
postSchema.pre("save", function (next) {
  if (this.isModified("title")) {
    this.slug = this.title
      .toLowerCase()
      .replace(/[^a-z0-9]+/g, "-")
      .replace(/^-|-$/g, "");
  }
  next();
});

// ── Types & Model ──────────────────────────────────────────
export type IPost  = InferSchemaType<typeof postSchema>;
export const Post: Model<IPost> = mongoose.model<IPost>("Post", postSchema);

// src/config/db.ts — typed MongoDB connection with retry logic
import mongoose from "mongoose";

const MAX_RETRIES = 5;
const RETRY_MS   = 3000;

export const connectDB = async (attempt = 1): Promise<void> => {
  try {
    const uri = process.env.MONGODB_URI;

    if (!uri) throw new Error("MONGODB_URI not set in environment");

    await mongoose.connect(uri, {
      dbName:          process.env.DB_NAME ?? "myapp",
      maxPoolSize:     10,          // connection pool
      serverSelectionTimeoutMS: 5000,
      socketTimeoutMS: 45000
    });

    console.log(`✓ MongoDB connected (attempt ${attempt})`);

  } catch (err) {
    console.error(`✗ MongoDB connection failed (attempt ${attempt})`, err);

    if (attempt < MAX_RETRIES) {
      console.log(`  Retrying in ${RETRY_MS / 1000}s…`);
      await new Promise(r => setTimeout(r, RETRY_MS));
      return connectDB(attempt + 1);
    }

    process.exit(1);
  }
};

// Graceful shutdown — close pool before process exits
process.on("SIGINT",  () => mongoose.connection.close());
process.on("SIGTERM", () => mongoose.connection.close());
