⌨️ 1. Debounced Search / Autocomplete
Debounced Search / Autocomplete Component
This single component tests controlled inputs (typing updates state), debouncing (don't fire an API call on every keystroke), async API calls inside useEffect, race conditions (an older, slower request shouldn't overwrite a newer result), and loading/empty/error states.

📋 What interviewers watch for

Do you debounce with a setTimeout + cleanup instead of a library? Do you cancel/ignore stale requests? Do you handle the empty-query case and show loading/no-results states?

DebouncedSearch.jsx 
import { useState, useEffect } from 'react';

function useDebouncedValue(value, delay = 400) {
  const [debounced, setDebounced] = useState(value);

  useEffect(() => {
    const timer = setTimeout(() => setDebounced(value), delay);
    return () => clearTimeout(timer); // cleanup — cancels the pending update on re-render
  }, [value, delay]);

  return debounced;
}

function DebouncedSearch() {
  const [query, setQuery] = useState('');
  const [results, setResults] = useState([]);
  const [loading, setLoading] = useState(false);
  const debouncedQuery = useDebouncedValue(query, 400);

  useEffect(() => {
    if (!debouncedQuery.trim()) {
      setResults([]);
      return;
    }

    let ignore = false; // race-condition guard
    setLoading(true);

    fetch(`/api/search?q=${debouncedQuery}`)
      .then(res => res.json())
      .then(data => {
        if (!ignore) { // only apply if this is still the latest request
          setResults(data);
          setLoading(false);
        }
      })
      .catch(() => !ignore && setLoading(false));

    return () => { ignore = true; }; // cancel effect from a previous keystroke
  }, [debouncedQuery]);

  return (
    <div>
      <input
        value={query}
        onChange={e => setQuery(e.target.value)}
        placeholder="Search..."
      />
      {loading && <p>Loading...</p>}
      {!loading && debouncedQuery && results.length === 0 && <p>No results</p>}
      <ul>
        {results.map(r => <li key={r.id}>{r.name}</li>)}
      </ul>
    </div>
  );
}
💡 Interview Answer: I separated the debouncing into a reusable useDebouncedValue hook, then fired the API call in a useEffect keyed on the debounced value. I used an ignore flag returned from the cleanup function to prevent a slow, stale response from overwriting a newer one — the classic race-condition bug in search boxes.
✅ 2. To-Do List App with Filters
To-Do List App (Full CRUD + Filters)
The baseline React test: Create, Read, Update, Delete on a list held in state, plus derived state for filtering (All / Active / Completed) without duplicating data, and correct use of stable, unique keys when rendering the list.

📋 What interviewers watch for

Do you mutate state directly (todos.push(...) — bad) or create new arrays/objects (good)? Do you filter with useMemo/plain derived logic instead of storing a second "filteredTodos" array that can drift out of sync? Do you use todo.id as the key, not the array index?

TodoApp.jsx 
import { useState } from 'react';

function TodoApp() {
  const [todos, setTodos] = useState([]);
  const [text, setText] = useState('');
  const [filter, setFilter] = useState('all'); // 'all' | 'active' | 'completed'

  // ── CREATE ──
  const addTodo = () => {
    if (!text.trim()) return;
    setTodos(prev => [...prev, { id: crypto.randomUUID(), text, done: false }]);
    setText('');
  };

  // ── UPDATE (toggle) ──
  const toggleTodo = (id) => {
    setTodos(prev => prev.map(t =>
      t.id === id ? { ...t, done: !t.done } : t
    ));
  };

  // ── UPDATE (edit text) ──
  const editTodo = (id, newText) => {
    setTodos(prev => prev.map(t => t.id === id ? { ...t, text: newText } : t));
  };

  // ── DELETE ──
  const deleteTodo = (id) => {
    setTodos(prev => prev.filter(t => t.id !== id));
  };

  // ── Derived state — NOT stored separately, always in sync ──
  const visibleTodos = todos.filter(t => {
    if (filter === 'active') return !t.done;
    if (filter === 'completed') return t.done;
    return true;
  });

  return (
    <div>
      <input value={text} onChange={e => setText(e.target.value)}
             onKeyDown={e => e.key === 'Enter' && addTodo()} />
      <button onClick={addTodo}>Add</button>

      <div>
        {['all', 'active', 'completed'].map(f => (
          <button key={f} onClick={() => setFilter(f)}
                  disabled={filter === f}>{f}</button
        ))}
      </div>

      <ul>
        {visibleTodos.map(todo => ( // key = todo.id, NOT array index
          <li key={todo.id}>
            <input type="checkbox" checked={todo.done} onChange={() => toggleTodo(todo.id)} />
            <span style={{ textDecoration: todo.done ? 'line-through' : 'none' }}>
              {todo.text}
            </span>
            <button onClick={() => deleteTodo(todo.id)}>🗑️</button>
          </li>
        ))}
      </ul>
    </div>
  );
}
💡 Interview Answer: I stored a single todos array as the source of truth, updated it immutably with map/filter/spread for create, update, and delete, and derived the visible list from filter state on every render instead of keeping a second synced array. Keys come from a stable id, never the array index, so React doesn't misidentify rows after a delete.
📜 3. Infinite Scroll List
Infinite Scroll List
Tests side effects (fetching more data as the user scrolls), refs (pointing at a DOM sentinel element), the IntersectionObserver API instead of scroll-event listeners, and awareness of performance (not re-fetching, not leaking observers) — a very common ask at feed-heavy product companies.

📋 What interviewers watch for

Do you use IntersectionObserver instead of a manual onScroll handler (which is less performant and harder to get right)? Do you clean up the observer in useEffect's return function? Do you guard against firing a duplicate request while one is already in flight?

InfiniteScrollList.jsx 
import { useState, useEffect, useRef, useCallback } from 'react';

function InfiniteScrollList() {
  const [items, setItems] = useState([]);
  const [page, setPage] = useState(1);
  const [loading, setLoading] = useState(false);
  const [hasMore, setHasMore] = useState(true);
  const sentinelRef = useRef(null);

  // ── Fetch a page whenever `page` changes ──
  useEffect(() => {
    let ignore = false;
    setLoading(true);

    fetch(`/api/items?page=${page}`)
      .then(res => res.json())
      .then(data => {
        if (ignore) return;
        setItems(prev => [...prev, ...data.items]);
        setHasMore(data.hasMore);
        setLoading(false);
      });

    return () => { ignore = true; };
  }, [page]);

  // ── Observe the sentinel div at the bottom of the list ──
  const loadMore = useCallback((entries) => {
    const target = entries[0];
    if (target.isIntersecting && !loading && hasMore) {
      setPage(p => p + 1);
    }
  }, [loading, hasMore]);

  useEffect(() => {
    const observer = new IntersectionObserver(loadMore, { rootMargin: '200px' });
    const el = sentinelRef.current;
    if (el) observer.observe(el);
    return () => observer.disconnect(); // avoid leaking observers on unmount
  }, [loadMore]);

  return (
    <div>
      <ul>
        {items.map(item => <li key={item.id}>{item.title}</li>)}
      </ul>
      {hasMore && <div ref={sentinelRef}>{loading ? 'Loading more...' : ''}</div>}
      {!hasMore && <p>You've reached the end.</p>}
    </div>
  );
}
💡 Interview Answer: I put a tiny sentinel div at the bottom of the list and watched it with IntersectionObserver instead of a scroll listener — it's cheaper and doesn't require throttling. The observer callback only bumps page when it's not already loading and there's more data, and the effect cleans up the observer to avoid leaks when the component unmounts.
🪟 4. Modal / Dialog Component from Scratch
Modal / Dialog Component (No UI Library)
A deceptively small component that packs in portals (rendering outside the parent DOM tree so overflow: hidden/z-index can't trap it), event listener cleanup (Esc key, outside click), and accessibility — a focus trap and returning focus to the trigger element on close.

📋 What interviewers watch for

Do you use createPortal instead of just absolutely-positioning a div inline? Do you remove the keydown listener in the cleanup function? Do you trap Tab focus inside the modal, and give it role="dialog" / aria-modal?

Modal.jsx 
import { useEffect, useRef } from 'react';
import { createPortal } from 'react-dom';

function Modal({ isOpen, onClose, children }) {
  const dialogRef = useRef(null);
  const triggerRef = useRef(document.activeElement); // remember what had focus

  // ── Esc key to close ──
  useEffect(() => {
    if (!isOpen) return;
    const handleKeyDown = (e) => { if (e.key === 'Escape') onClose(); };
    document.addEventListener('keydown', handleKeyDown);
    return () => document.removeEventListener('keydown', handleKeyDown);
  }, [isOpen, onClose]);

  // ── Focus trap + move focus in/out ──
  useEffect(() => {
    if (!isOpen) return;
    dialogRef.current?.focus();

    const trapTab = (e) => {
      if (e.key !== 'Tab') return;
      const focusable = dialogRef.current.querySelectorAll(
        'button, [href], input, select, textarea, [tabindex]:not([tabindex="-1"])'
      );
      const first = focusable[0], last = focusable[focusable.length - 1];
      if (e.shiftKey && document.activeElement === first) { e.preventDefault(); last.focus(); }
      else if (!e.shiftKey && document.activeElement === last) { e.preventDefault(); first.focus(); }
    };

    document.addEventListener('keydown', trapTab);
    return () => {
      document.removeEventListener('keydown', trapTab);
      triggerRef.current?.focus(); // return focus to trigger on close
    };
  }, [isOpen]);

  if (!isOpen) return null;

  return createPortal( // escapes any parent overflow/z-index issues
    <div className="overlay" onClick={onClose}>
      <div
        ref={dialogRef}
        role="dialog"
        aria-modal="true"
        tabIndex={-1}
        className="dialog"
        onClick={e => e.stopPropagation()} // don't close when clicking inside
      >
        {children}
        <button onClick={onClose}>Close</button>
      </div>
    </div>,
    document.getElementById('modal-root') // a div outside #root in index.html
  );
}
💡 Interview Answer: I rendered the dialog through createPortal into a dedicated #modal-root so it's never clipped by an ancestor's overflow: hidden. Two effects handle the Esc key and a manual Tab focus trap, both fully cleaned up when the modal closes or unmounts, and focus returns to whatever triggered the modal — the accessibility detail most people forget.
🔢 5. Pagination Component (with Sorting/Filtering)
Pagination Component (+ Sortable, Filterable Table)
Very common at companies building admin dashboards or data-heavy products. Tests derived state (current page slice, computed from data + page + sort + filter), resetting page on filter/sort change, and cleanly composing multiple pieces of interacting state without bugs.

📋 What interviewers watch for

Do you reset to page 1 when the filter or sort changes (a very common missed edge case — otherwise users land on an empty page)? Do you compute the visible slice with useMemo instead of re-filtering on every unrelated render? Do you guard page bounds (e.g. deleting the last item on the last page)?

PaginatedTable.jsx 
import { useState, useMemo } from 'react';

const PAGE_SIZE = 10;

function PaginatedTable({ rows }) {
  const [page, setPage] = useState(1);
  const [sortKey, setSortKey] = useState('name');
  const [filterText, setFilterText] = useState('');

  // ✅ derived, not duplicated: filter → sort → paginate
  const filteredSorted = useMemo(() => {
    return rows
      .filter(r => r.name.toLowerCase().includes(filterText.toLowerCase()))
      .sort((a, b) => (a[sortKey] > b[sortKey] ? 1 : -1));
  }, [rows, filterText, sortKey]);

  const totalPages = Math.max(1, Math.ceil(filteredSorted.length / PAGE_SIZE));
  // ✅ clamp page — handles filter shrinking the result set below current page
  const safePage = Math.min(page, totalPages);

  const visibleRows = filteredSorted.slice(
    (safePage - 1) * PAGE_SIZE,
    safePage * PAGE_SIZE
  );

  // ✅ reset to page 1 whenever filter or sort changes
  const handleFilterChange = (value) => {
    setFilterText(value);
    setPage(1);
  };
  const handleSortChange = (key) => {
    setSortKey(key);
    setPage(1);
  };

  return (
    <div>
      <input placeholder="Filter by name..." value={filterText}
             onChange={e => handleFilterChange(e.target.value)} />
      <select value={sortKey} onChange={e => handleSortChange(e.target.value)}>
        <option value="name">Name</option>
        <option value="date">Date</option>
      </select>

      <table>
        <tbody>
          {visibleRows.map(row => (
            <tr key={row.id}><td>{row.name}</td><td>{row.date}</td></tr>
          ))}
        </tbody>
      </table>

      <div>
        <button disabled={safePage === 1} onClick={() => setPage(p => p - 1)}>Prev</button>
        <span>Page {safePage} of {totalPages}</span>
        <button disabled={safePage === totalPages} onClick={() => setPage(p => p + 1)}>Next</button>
      </div>
    </div>
  );
}
💡 Interview Answer: I derived filteredSorted from the raw rows with useMemo instead of storing a separate filtered array, then sliced it for the current page. Changing the filter or sort resets page back to 1, and I clamp safePage to totalPages so a shrinking result set (e.g. after filtering or deleting a row) never leaves the user stranded on a blank page.
