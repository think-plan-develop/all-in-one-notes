⌨️ 1. Debouncing
src/debouncing/App.jsx
Typing into the input updates text immediately, but the fetch call is delayed with setTimeout and reset via clearTimeout on every keystroke — so a network request only fires once the user pauses for 500ms.

📋 Note in the code

The typing timer id lives outside the component. Declared inside App, it would be recreated every render and the reference to clear would be lost; declared at module scope it persists across renders — but is then shared by every instance of App, which is worth calling out as a trade-off.

App.jsx
import { useState } from "react";
import "./App.css";

let typing;

function App() {
  const [text, setText] = useState("1");
  const [user, setUser] = useState({});

  const handleChange = (event) => {
    const value = event.target.value.slice(-1);
    setText(value);
    clearTimeout(typing);

    typing = setTimeout(() => {
      fetch(`https://jsonplaceholder.typicode.com/todos/${value}`)
        .then((res) => res.json())
        .then((data) => setUser(data));
    }, 500);
  };

  return (
    <div className="debounce-demo">
      <div className="debounce-intro">
        <span className="debounce-label">debounce : Pause for 500ms</span>
      </div>

      <input
        className="debounce-control"
        type="number"
        value={text}
        onChange={handleChange}
        aria-label="Todo ID"
      />

      <pre className="debounce-result">{JSON.stringify(user, null, 2)}</pre>
    </div>
  );
}

export default App;
⚠️ Gotcha to remember: because typing is a bare module-level variable rather than a ref, this pattern only works safely for a single mounted instance of the component — two instances on screen at once would fight over the same timer.

🎨 App.css highlights

The workspace (.debounce-demo) is a soft gradient card; the input (.debounce-control) is styled dark with a teal border that shifts to orange on focus and lifts slightly (translateY(-2px)); the result (.debounce-result) is a scrollable monospace terminal-style block for the raw JSON response.

✅ 2. Todo List
src/todo/App.jsx
A single todoList array in state supports full CRUD: add a trimmed task, edit by loading a todo back into the input and tracking its id in editId, and delete by filtering it out — with a useRef that refocuses the input the moment editing starts.

📋 Note in the code

Submitting the form branches on whether editId is set: if it is, map swaps in the updated name for the matching todo; otherwise a new todo is appended with a fresh crypto.randomUUID() key. Deleting the todo currently being edited also clears the edit state so the form doesn't point at a task that no longer exists.

App.jsx
import { useRef, useState } from "react";
import "./App.css";

function App() {
  const [todoList, setTodoList] = useState([]);
  const [task, setTask] = useState("");
  const [editId, setEditId] = useState(null);
  const inputRef = useRef(null);

  const handleSubmit = (event) => {
    event.preventDefault();
    const trimmedTask = task.trim();
    if (!trimmedTask) return;

    if (editId) {
      setTodoList((current) => current.map((todo) => (
        todo.id === editId ? { ...todo, name: trimmedTask } : todo
      )));
      setEditId(null);
    } else {
      setTodoList((current) => [
        ...current,
        { id: crypto.randomUUID(), name: trimmedTask },
      ]);
    }

    setTask("");
  };

  const editTodo = (todo) => {
    setEditId(todo.id);
    setTask(todo.name);
    inputRef.current?.focus();
  };

  const deleteTodo = (id) => {
    setTodoList((current) => current.filter((todo) => todo.id !== id));
    if (editId === id) {
      setEditId(null);
      setTask("");
    }
  };

  return (
    <section className="todo-card">
      <div className="section-kicker">03 / Local state</div>
      <h2>Todo list</h2>
      <p className="section-description">Add, edit, and remove tasks in memory.</p>
      <form className="todo-form" onSubmit={handleSubmit}>
        <input
          ref={inputRef}
          className="todo-input"
          type="text"
          placeholder="Enter a task"
          value={task}
          onChange={(event) => setTask(event.target.value)}
        />
        <button className="primary-button" type="submit">{editId ? "Update" : "Add"}</button>
      </form>
      {editId && (
        <button className="cancel-button" type="button" onClick={() => { setEditId(null); setTask(""); }}>
          Cancel editing
        </button>
      )}
      {todoList.length > 0 && (
        <div className="todo-table-wrapper">
          <table className="todo-table">
            <thead>
              <tr><th>#</th><th>Task</th><th>Edit</th><th>Delete</th></tr>
            </thead>
            <tbody>
              {todoList.map((todo, index) => (
                <tr key={todo.id}>
                  <td className="todo-number">{String(index + 1).padStart(2, "0")}</td>
                  <td className="todo-name">{todo.name}</td>
                  <td><button onClick={() => editTodo(todo)}>Edit</button></td>
                  <td><button onClick={() => deleteTodo(todo.id)}>Delete</button></td>
                </tr>
              ))}
            </tbody>
          </table>
        </div>
      )}
      {todoList.length === 0 && <p className="empty-message">Your list is ready for its first task.</p>}
    </section>
  );
}

export default App;
💡 Design detail: keys use todo.id (a UUID), never the array index — so rows stay correctly matched to their data even after a delete shifts positions.
🪟 3. Modal
src/modal/App.jsx
A single showModal boolean gates the overlay: clicking "Open modal" flips it to true, rendering a fixed, dark overlay with a centered white panel; the x close button flips it back to false.
App.jsx
import { useState } from "react";
import "./App.css";

function App() {
  const [showModal, setShowModal] = useState(false);

  return (
    <section className="modal-card">
      <div className="section-kicker">02 / Focused interaction</div>
      <h2>Modal</h2>

      <button className="primary-button open-button" onClick={() => setShowModal(true)}>
        Open modal
      </button>

      {showModal && (
        <div className="overlay">
          <div className="modal">
            <button className="close" onClick={() => setShowModal(false)}>x</button>
            <h2>Modal</h2>
            <input type="text" placeholder="Enter name" />
            <button className="submit">Submit</button>
          </div>
        </div>
      )}
    </section>
  );
}

export default App;
📝 What's missing vs. a production modal: no portal (createPortal), no Esc-key listener, no focus trap, and no click-outside-to-close — this version relies purely on the explicit close button and conditional rendering, which is the simplest possible baseline.

🎨 App.css highlights

The .overlay is a fixed, full-screen flex container with a semi-transparent black background that centers the .modal panel; the panel itself is a fixed-width white card with the close button absolutely positioned in its top-right corner.
