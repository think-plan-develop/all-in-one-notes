Login.test.jsx
// ---------------------------------------------------------
// Import Testing Library Config Elements
// ---------------------------------------------------------

import { render, screen, waitFor } from "@testing-library/react"; // Render component, query DOM, wait for updates
import userEvent from "@testing-library/user-event";              // Simulate real user actions (click, type)
import Login from "../Login";                                     // Component under test

// ---------------------------------------------------------
// React Router Environment Mocking
// ---------------------------------------------------------

import { MemoryRouter } from "react-router-dom";                  // Router used only for testing
const mockNavigate = jest.fn();                                   // Mock function to verify navigation

// Replace only useNavigate with mock handler
jest.mock("react-router-dom", () => ({
  ...jest.requireActual("react-router-dom"),                      // Keep all original router functionality
  useNavigate: () => mockNavigate,                                // Replace useNavigate() hook tracking
}));

// ---------------------------------------------------------
// Redux Global Store Connection
// ---------------------------------------------------------

import { Provider } from "react-redux";
import { store } from "../redux/store";                           // Real or mock store pipeline

// ---------------------------------------------------------
// API Architecture Mocking Setup
// ---------------------------------------------------------

import { loginUser } from "../services/api";

// Replace actual structural API network with a mock double
jest.mock("../services/api", () => ({
  loginUser: jest.fn(),
}));

// ---------------------------------------------------------
// Core Jest Execution Lifecycle Engine
// ---------------------------------------------------------

describe("Login Component", () => {

  beforeEach(() => {
    jest.clearAllMocks();                                         // Clear mock history before every test cycle
  });

  test("should login successfully", async () => {

    // STEP 1: Arrange (Prepare Mocked Responses and Data Structures)
    loginUser.mockResolvedValue({
      data: {
        token: "123456",
        user: {
          name: "Shivam"
        }
      }
    });

    // STEP 2: Render Component Tree with Wrapped Testing Providers
    render(
      <Provider store={store}>                                 {/* Connect Redux Container */}
        <MemoryRouter>                                          {/* Connect Router Sandbox */}
          <Login />
        </MemoryRouter>
      </Provider>
    );

    // STEP 3: Query DOM Nodes and Active Viewport Controls
    const emailInput = screen.getByPlaceholderText("Enter Email");
    const passwordInput = screen.getByPlaceholderText("Enter Password");
    const loginButton = screen.getByRole("button", { name: /login/i });

    // STEP 4: Execute Interactive User Micro-Events via Promises
    await userEvent.type(emailInput, "admin@gmail.com");
    await userEvent.type(passwordInput, "123456");

    // CHECKPOINT A: Verify State-Driven Input Node Value Updates
    expect(emailInput).toHaveValue("admin@gmail.com");
    expect(passwordInput).toHaveValue("123456");

    // STEP 5: Dispatch Form Submit Trigger Event
    await userEvent.click(loginButton);

    // CHECKPOINT B: Verify Core API Metrics and Request Interceptions
    expect(loginUser).toHaveBeenCalledTimes(1);
    expect(loginUser).toHaveBeenCalledWith({
      email: "admin@gmail.com",
      password: "123456",
    });

    // CHECKPOINT C: Await Dynamic DOM Reconciliation Updates
    await waitFor(() => {
      expect(screen.getByText("Welcome Shivam")).toBeInTheDocument();
    });

    // CHECKPOINT D: Track App-Level Navigation Redirect Interception
    expect(mockNavigate).toHaveBeenCalledWith("/dashboard");

  });

});
