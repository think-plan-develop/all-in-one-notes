    "abcabcbb"
    Show Answer 
    function longestSubstring(str) {
  let start = 0, maxLen = 0;
  const seen = new Map();

  for (let end = 0; end < str.length; end++) {
    const ch = str[end];
    if (seen.has(ch) && seen.get(ch) >= start) {
      start = seen.get(ch) + 1;
    }
    seen.set(ch, end);
    maxLen = Math.max(maxLen, end - start + 1);
  }
  return maxLen;
}

🔴 Part 3 — Advanced JavaScript (15 Questions, Fully Explained)
1. Implement debounce() ⭐⭐⭐⭐⭐

What it does

Delays invoking a function until after a specified wait time has elapsed since the last time it was invoked. Used for search inputs, resize handlers, etc. — so the function only fires once the user "settles."

debounce.js
function debounce(fn, delay) {
  let timer;
  return function (...args) {
    clearTimeout(timer);
    timer = setTimeout(() => fn.apply(this, args), delay);
  };
}

// Usage
const handleSearch = debounce((query) => {
  console.log("Searching for:", query);
}, 300);

input.addEventListener("input", (e) => handleSearch(e.target.value));
Key idea: every new call resets the timer, so the function only runs after calls stop for delay ms.
2. Implement throttle() ⭐⭐⭐⭐⭐

What it does

Ensures a function runs at most once every specified interval, regardless of how many times it's triggered. Used for scroll or mousemove handlers.

throttle.js
function throttle(fn, limit) {
  let inThrottle = false;
  return function (...args) {
    if (!inThrottle) {
      fn.apply(this, args);
      inThrottle = true;
      setTimeout(() => (inThrottle = false), limit);
    }
  };
}

// Usage
window.addEventListener("scroll", throttle(() => {
  console.log("Scroll event handled");
}, 200));
Debounce waits for silence; throttle guarantees a steady drip — at most one call per interval.
3. Implement Currying ⭐⭐⭐⭐

What it does

Transforms a function that takes multiple arguments into a sequence of functions, each taking a single argument (or a subset), returning a new function until all arguments are supplied.

curry.js
function curry(fn) {
  return function curried(...args) {
    if (args.length >= fn.length) {
      return fn.apply(this, args);
    }
    return (...next) => curried.apply(this, [...args, ...next]);
  };
}

// Usage
function sum(a, b, c) { return a + b + c; }
const curriedSum = curry(sum);

curriedSum(1)(2)(3);   // 6
curriedSum(1, 2)(3);    // 6
curriedSum(1, 2, 3);     // 6
4. Implement Memoization ⭐⭐⭐⭐

What it does

Wraps a function so repeated calls with the same arguments return a cached result instead of recomputing — useful for expensive/pure computations.

memoize.js
function memoize(fn) {
  const cache = new Map();
  return function (...args) {
    const key = JSON.stringify(args);
    if (cache.has(key)) return cache.get(key);
    const result = fn.apply(this, args);
    cache.set(key, result);
    return result;
  };
}

// Usage
const slowSquare = (n) => { for(let i=0;i<1e8;i++){} return n*n; };
const fastSquare = memoize(slowSquare);

fastSquare(5); // slow first time
fastSquare(5); // instant (cached)

11. Implement Promise.all() (Polyfill) ⭐⭐⭐⭐⭐

What it does

Takes an array of promises and resolves when all resolve (preserving order), or rejects as soon as any one rejects.

promise-all.js
function promiseAll(promises) {
  return new Promise((resolve, reject) => {
    const results = [];
    let completed = 0;

    if (promises.length === 0) resolve(results);

    promises.forEach((p, i) => {
      Promise.resolve(p)
        .then((value) => {
          results[i] = value;
          completed++;
          if (completed === promises.length) resolve(results);
        })
        .catch(reject);
    });
  });
}
12. Implement Promise.allSettled() (Polyfill) ⭐⭐⭐⭐

What it does

Waits for all promises to settle (either resolve or reject) and returns a status/value or reason for each — never short-circuits on rejection.

promise-allsettled.js
function promiseAllSettled(promises) {
  return new Promise((resolve) => {
    const results = [];
    let completed = 0;

    if (promises.length === 0) resolve(results);

    promises.forEach((p, i) => {
      Promise.resolve(p)
        .then((value) => { results[i] = { status: "fulfilled", value }; })
        .catch((reason) => { results[i] = { status: "rejected", reason }; })
        .finally(() => {
          completed++;
          if (completed === promises.length) resolve(results);
        });
    });
  });
}
13. Implement Promise.race() ⭐⭐⭐⭐

What it does

Resolves or rejects as soon as the first promise in the array settles — whichever happens first, win or lose.

promise-race.js
function promiseRace(promises) {
  return new Promise((resolve, reject) => {
    promises.forEach((p) => {
      Promise.resolve(p).then(resolve).catch(reject);
    });
  });
}
14. Retry Failed API Requests ⭐⭐⭐⭐

What it does

Automatically retries a failing async operation (e.g. a fetch call) up to N times, often with a delay between attempts, before finally giving up.

retry.js
async function retryRequest(fn, retries = 3, delay = 1000) {
  try {
    return await fn();
  } catch (err) {
    if (retries === 0) throw err;
    console.log(`Retrying... attempts left: ${retries}`);
    await new Promise((res) => setTimeout(res, delay));
    return retryRequest(fn, retries - 1, delay);
  }
}

// Usage
retryRequest(() => fetch("/api/data").then(r => r.json()));
