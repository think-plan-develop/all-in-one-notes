    function findDuplicates(arr) {
  const seen = new Set();
  const dupes = new Set();
  for (const n of arr) {
    if (seen.has(n)) dupes.add(n);
    seen.add(n);
  }
  return [...dupes];
}

  • 8 Merge two sorted arrays ⭐

    Show Answer 
    function mergeSorted(a, b) {
  const result = [];
  let i = 0, j = 0;
  while (i < a.length && j < b.length) {
    result.push(a[i] <= b[j] ? a[i++] : b[j++]);
  }
  return [...result, ...a.slice(i), ...b.slice(j)];
}

  • 14 Find the maximum sum subarray (Kadane's Algorithm) ⭐

    Show Answer 
    function maxSubArray(arr) {
  let maxSoFar = arr[0];
  let maxEndingHere = arr[0];

  for (let i = 1; i < arr.length; i++) {
    maxEndingHere = Math.max(arr[i], maxEndingHere + arr[i]);
    maxSoFar = Math.max(maxSoFar, maxEndingHere);
  }
  return maxSoFar;
}

  • 15 Sort an array without using built-in sort()

    function sortArray(arr) {
  for (let i = 0; i < arr.length; i++) {
    for (let j = 0; j < arr.length - i - 1; j++) {
      if (arr[j] > arr[j + 1]) {
        [arr[j], arr[j + 1]] = [arr[j + 1], arr[j]];
      }
    }
  }
  return arr;
}

  • 20 Find elements that occur more than once

    function findDuplicates(arr) {
  const frequency = {};
  const duplicates = [];
  for (let item of arr) {
    frequency[item] = (frequency[item] || 0) + 1;
  }
  for (let item in frequency) {
    if (frequency[item] > 1) {
      duplicates.push(item);
    }
  }
  return duplicates;
}

    // Cleaner approach using Set:
function findDuplicates(arr) {
  const seen = new Set();
  const duplicates = new Set();
  for (let item of arr) {
    if (seen.has(item)) {
      duplicates.add(item);
    } else {
      seen.add(item);
    }
  }
  return [...duplicates];
}

findDuplicates([1, 2, 3, 2, 4, 1]); // [2, 1]

    // Manual map()
function myMap(arr, callback) {

    // Manual filter()
function myFilter(arr, callback) {
  const result = [];
  for (let i = 0; i < arr.length; i++) {
    if (callback(arr[i], i, arr)) {
      result.push(arr[i]);
    }
  }
  return result;
}

myFilter([1, 2, 3, 4, 5], num => num % 2 === 0); // [2, 4]
    // Manual reduce()
function myReduce(arr, callback, initialValue) {
  let accumulator = initialValue;
  for (let i = 0; i < arr.length; i++) {
    accumulator = callback(accumulator, arr[i], i, arr);
  }
  return accumulator;
}

myReduce([1, 2, 3, 4], (acc, num) => acc + num, 0); // 10
