    function isPalindrome(str) {
  const clean = str.toLowerCase().replace(/[^a-z0-9]/g, "");
  return clean === clean.split("").reverse().join("");
}

isPalindrome("madam"); // true

    function isAnagram(a, b) {
  const normalize = (s) => s.toLowerCase().split("").sort().join("");
  return normalize(a) === normalize(b);
}

isAnagram("listen", "silent"); // true
  • 4 Find the first non-repeating character
    "swiss"
    Show Answer 
    function firstNonRepeating(str) {
  const counts = {};
  for (const ch of str) counts[ch] = (counts[ch] || 0) + 1;
  for (const ch of str) if (counts[ch] === 1) return ch;
  return null;
}

firstNonRepeating("swiss"); // "w"

  • 6 Count vowels and consonants
    "Interview Preparation"
    Show Answer 
    function countVowelsAndConsonants(str) {
  const letters = str.replace(/[^a-z]/gi, "");
  const vowelMatches = letters.match(/[aeiou]/gi);
  const vowels = vowelMatches ? vowelMatches.length : 0;
  const consonants = letters.length - vowels;
  return { vowels, consonants };
}

countVowelsAndConsonants("Interview Preparation");
// { vowels: 9, consonants: 11 }
  • 7 Find duplicate characters New
    "programming"
    Show Answer 
    function findDuplicateChars(str) {
  const seen = new Set();
  const duplicates = new Set();

  for (const ch of str.toLowerCase()) {
    if (ch === " ") continue;
    if (seen.has(ch)) duplicates.add(ch);
    seen.add(ch);
  }
  return [...duplicates];
}

findDuplicateChars("programming"); // ["r", "g", "m"]

  • 9 Find the longest word in a string New
    "The quick brown fox jumps"
    Show Answer 
    function longestWord(str) {
  const words = str.split(" ");
  return words.reduce((longest, current) =>
    current.length > longest.length ? current : longest, "");
}

longestWord("The quick brown fox jumps"); // "quick"

    " The quick brown fox jumps "
    Show Answer 
    function countWords(str) {
  const trimmed = str.trim();
  if (trimmed === "") return 0;
  return trimmed.split(/\s+/).length;
}

countWords("  The quick brown   fox jumps  "); // 5

  • 15 Valid Parentheses ⭐ Important
    "({[]})"
    Show Answer 
    function isValidParentheses(str) {
  const stack = [];
  const pairs = { ')': '(', '}': '{', ']': '[' };

  for (let char of str) {
    if (char in pairs) {
      if (stack.pop() !== pairs[char]) {
        return false;
      }
    } else {
      stack.push(char);
    }
  }

  return stack.length === 0;
}

console.log(isValidParentheses('({[]})')); // Output: true
